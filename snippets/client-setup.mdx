```java Java
import com.webcommander.client.ClientRegistry;
import com.webcommander.config.AuthConfig;
import com.webcommander.config.SdkConfig;
import java.util.List;

public class Main {
    public static void main(String[] args) throws Exception {
        AuthConfig auth = AuthConfig.builder()
                .baseUrl(System.getenv("WC_BASE_URL"))
                .clientId(System.getenv("WC_CLIENT_ID"))
                .clientSecret(System.getenv("WC_CLIENT_SECRET"))
                .authBaseUrl(System.getenv("WC_AUTH_BASE_URL"))
                .clientName("default")
                .build();
        SdkConfig config = SdkConfig.builder(auth).raiseOnError(true).build();
        ClientRegistry.registerInstances(List.of(config));
    }
}
```

```go Go
package main

import (
	"log"
	"os"

	wc "github.com/WebCommanderCMS/webcommander-sdk-go/v4"
)

func main() {
	err := wc.RegisterInstances(wc.SdkConfig{
		Auth: wc.AuthConfig{
			ClientName:   "default",
			BaseURL:      os.Getenv("WC_BASE_URL"),
			ClientID:     os.Getenv("WC_CLIENT_ID"),
			ClientSecret: os.Getenv("WC_CLIENT_SECRET"),
			AuthBaseURL:  os.Getenv("WC_AUTH_BASE_URL"),
		},
		RaiseOnError: true,
	})
	if err != nil {
		log.Fatal(err)
	}
}
```
