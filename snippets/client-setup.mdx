```python Python
import os

from webcommander import AuthConfig, build_client, register_instances

register_instances([
    AuthConfig(
        client_name="default",
        base_url=os.environ["WC_BASE_URL"],
        client_id=os.environ["WC_CLIENT_ID"],
        client_secret=os.environ["WC_CLIENT_SECRET"],
        redirect_uri=os.environ["WC_REDIRECT_URI"],
        auth_string=os.environ["WC_AUTH_STRING"],
    ),
])

client = build_client(raise_on_error=True)
```

```go Go
import (
	"context"
	"fmt"
	"log"
	"os"

	wc "bitbucket.org/webalive/webcommander-sdk-go/v3"
)

ctx := context.Background()
client, err := wc.NewClient(wc.SdkConfig{
	Auth: wc.AuthConfig{
		ClientName:   "default",
		BaseURL:      os.Getenv("WC_BASE_URL"),
		ClientID:     os.Getenv("WC_CLIENT_ID"),
		ClientSecret: os.Getenv("WC_CLIENT_SECRET"),
		RedirectURI:  os.Getenv("WC_REDIRECT_URI"),
		AuthString:   os.Getenv("WC_AUTH_STRING"),
	},
	// Return a failed HTTP call as an error instead of a response with Succeeded() false.
	RaiseOnError: true,
})
if err != nil {
	log.Fatal(err)
}
defer client.Close()
```
