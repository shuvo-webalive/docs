```java Java
import com.webcommander.client.ClientRegistry;
import com.webcommander.client.WebCommanderClient;
import com.webcommander.config.AuthConfig;
import com.webcommander.config.SdkConfig;
import java.util.List;

public class Main {
    public static void main(String[] args) throws Exception {
        AuthConfig auth = AuthConfig.builder()
                .baseUrl(System.getenv("WC_BASE_URL"))
                .clientId(System.getenv("WC_CLIENT_ID"))
                .clientSecret(System.getenv("WC_CLIENT_SECRET"))
                .authBaseUrl(System.getenv("WC_AUTH_BASE_URL"))
                .clientName("default")
                .build();
        AuthConfig legacyAuth = AuthConfig.builder()
                .baseUrl(System.getenv("WC_LEGACY_BASE_URL"))
                .clientId(System.getenv("WC_LEGACY_CLIENT_ID"))
                .clientSecret(System.getenv("WC_LEGACY_CLIENT_SECRET"))
                .redirectUri(System.getenv("WC_LEGACY_REDIRECT_URI"))
                .authString(System.getenv("WC_LEGACY_AUTH_STRING"))
                .clientName("legacy")
                .build();
        ClientRegistry.registerInstances(List.of(
                SdkConfig.builder(auth).raiseOnError(true).build(),
                SdkConfig.builder(legacyAuth).raiseOnError(true).build()));

        WebCommanderClient client = ClientRegistry.buildClient();
        WebCommanderClient legacyClient = ClientRegistry.buildClient("legacy");
    }
}
```

```go Go
package main

import (
	"log"
	"os"

	wc "github.com/WebCommanderCMS/webcommander-sdk-go/v4"
)

func main() {
	err := wc.RegisterInstances(
		wc.SdkConfig{
			Auth: wc.AuthConfig{
				ClientName:   "default",
				BaseURL:      os.Getenv("WC_BASE_URL"),
				ClientID:     os.Getenv("WC_CLIENT_ID"),
				ClientSecret: os.Getenv("WC_CLIENT_SECRET"),
				AuthBaseURL:  os.Getenv("WC_AUTH_BASE_URL"),
			},
			RaiseOnError: true,
		},
		wc.SdkConfig{
			Auth: wc.AuthConfig{
				ClientName:   "legacy",
				BaseURL:      os.Getenv("WC_LEGACY_BASE_URL"),
				ClientID:     os.Getenv("WC_LEGACY_CLIENT_ID"),
				ClientSecret: os.Getenv("WC_LEGACY_CLIENT_SECRET"),
				RedirectURI:  os.Getenv("WC_LEGACY_REDIRECT_URI"),
				AuthString:   os.Getenv("WC_LEGACY_AUTH_STRING"),
			},
			RaiseOnError: true,
		},
	)
	if err != nil {
		log.Fatal(err)
	}

	client, err := wc.BuildClient(nil)
	if err != nil {
		log.Fatal(err)
	}
	defer client.Close()

	legacyClient, err := wc.BuildClient("legacy")
	if err != nil {
		log.Fatal(err)
	}
	defer legacyClient.Close()
}
```
